import pandas as pd

# Load the CSV file
file_path = "/mnt/data/strain_sensitivity_v6_withsitechar.csv"
df = pd.read_csv(file_path)

# Display basic information and the first few rows of the dataset
df.info(), df.head()

import matplotlib.pyplot as plt
import seaborn as sns

# Drop rows with missing values in relevant columns
df_clean = df.dropna(subset=["Profile_name", "d max", "strain max 1", "Failure stage"])

# Sort Profile_name based on Failure stage
profile_order = df_clean.groupby("Profile_name")["Failure stage"].min().sort_values().index

# Create the figure and axes
fig, axes = plt.subplots(2, 1, figsize=(12, 10), sharex=True)

# Upper plot: d max
sns.boxplot(data=df_clean, x="Profile_name", y="d max", order=profile_order, ax=axes[0])
axes[0].set_title("Distribution of d max by Profile Name")
axes[0].set_ylabel("d max")

# Lower plot: epsilon max (strain max 1)
sns.boxplot(data=df_clean, x="Profile_name", y="strain max 1", order=profile_order, ax=axes[1])
axes[1].set_title("Distribution of Epsilon max (Strain max 1) by Profile Name")
axes[1].set_ylabel("Epsilon max")

# Rotate x-axis labels for better readability
plt.xticks(rotation=90)
plt.xlabel("Profile Name")

# Adjust layout
plt.tight_layout()
plt.show()
